import { useState } from 'react';
import { motion } from 'framer-motion';
import toast from 'react-hot-toast';
import {
  Zap, Check, Lock, ToggleLeft, ToggleRight, Users, Clock, BarChart3,
  BookOpen, Phone, Mic, Globe, Code, FileText, Target, Trophy, Brain,
  Shield, ChevronRight, Star,
} from 'lucide-react';
import { usePlanStore, useAuthStore } from '@/lib/store';
import { PlanTier, PLAN_CONFIGS, PlanFeatures } from '@/types';
import clsx from 'clsx';

type FeatureKey = keyof PlanFeatures;

interface ModuleRow {
  key: FeatureKey;
  label: string;
  description: string;
  icon: React.ElementType;
  capKey?: 'sessionsPerMonth' | 'agentsMax' | 'sessionMinutesMax';
}

const MODULES: ModuleRow[] = [
  { key: 'knowledgeBase',       label: 'Knowledge Base',       description: 'Train the AI bot with docs, URLs, and text content', icon: BookOpen },
  { key: 'preCallBriefing',     label: 'Pre-Call Briefing',    description: 'Share product materials with users before each call', icon: FileText },
  { key: 'customPersonas',      label: 'Custom Personas',      description: 'Create and save custom AI personas', icon: Brain },
  { key: 'teamRoleplays',       label: 'Team Roleplays',       description: 'Assign scenarios to agents from a shared library', icon: Users },
  { key: 'analytics',           label: 'Analytics',            description: 'Talk ratio, pacing, filler words, and monologue stats', icon: BarChart3 },
  { key: 'leaderboard',         label: 'Leaderboard',          description: 'Team performance rankings', icon: Trophy },
  { key: 'recordings',          label: 'Call Recordings',      description: 'Record and replay practice sessions', icon: Mic },
  { key: 'aiCoaching',          label: 'AI Coaching',          description: 'Automated per-criterion coaching tips and feedback', icon: Target },
  { key: 'evaluationPrompts',   label: 'Evaluation Prompts',   description: 'Customise the AI scoring rubric per roleplay type', icon: Shield },
  { key: 'multiLanguage',       label: 'Multi-Language',       description: 'Practice in 11 languages including Hindi, Spanish, French', icon: Globe },
  { key: 'apiAccess',           label: 'API Access',           description: 'Programmatic access to sessions and analytics data', icon: Code },
];

const PLAN_ORDER: PlanTier[] = ['starter', 'growth', 'pro', 'enterprise'];

function PlanBadge({ plan }: { plan: PlanTier }) {
  const cfg = PLAN_CONFIGS[plan];
  return (
    <span
      className="inline-flex items-center gap-1 px-2 py-0.5 rounded-full text-[10px] font-bold uppercase tracking-wide"
      style={{ background: `${cfg.color}18`, color: cfg.color, border: `1px solid ${cfg.color}30` }}
    >
      {cfg.label}
    </span>
  );
}

function CapBadge({ label, value }: { label: string; value: number | null }) {
  return (
    <span className="text-[10px] px-1.5 py-0.5 rounded font-mono" style={{ background: 'var(--bg3)', color: 'var(--text3)', border: '1px solid var(--border)' }}>
      {label}: {value === null ? '∞' : value}
    </span>
  );
}

export function PlanSettingsPage() {
  const user = useAuthStore(s => s.user);
  const { plan, setPlan, moduleOverrides, setModuleOverride, getFeatures } = usePlanStore();
  const isAdmin = user?.role && ['COMPANY_ADMIN', 'SUPER_ADMIN'].includes(user.role);
  const features = getFeatures();

  const [saving, setSaving] = useState(false);

  const handlePlanChange = (p: PlanTier) => {
    if (!isAdmin) return;
    setPlan(p);
    toast.success(`Plan changed to ${PLAN_CONFIGS[p].label}`);
  };

  const handleToggle = (key: FeatureKey) => {
    if (!isAdmin) return;
    const current = features[key] as boolean;
    setModuleOverride(key, !current);
    toast.success(`${!current ? 'Enabled' : 'Disabled'} — override saved`);
  };

  const clearOverride = (key: FeatureKey) => {
    if (!isAdmin) return;
    const overrides = { ...usePlanStore.getState().moduleOverrides };
    delete overrides[key];
    usePlanStore.setState({ moduleOverrides: overrides });
    toast.success('Override cleared — using plan default');
  };

  const hasOverride = (key: FeatureKey) => key in moduleOverrides;
  const planDefault = (key: FeatureKey) => PLAN_CONFIGS[plan].features[key] as boolean;

  return (
    <div className="flex flex-col gap-6 max-w-4xl">

      {/* Header */}
      <div>
        <h2 className="font-display text-xl font-bold" style={{ color: 'var(--text)' }}>Plan & Modules</h2>
        <p className="text-sm mt-0.5" style={{ color: 'var(--text3)' }}>
          Manage your subscription plan and toggle individual feature modules for your organisation.
        </p>
      </div>

      {/* Plan selector */}
      <div className="rounded-[14px] border p-5 flex flex-col gap-4" style={{ background: 'var(--bg2)', borderColor: 'var(--border)' }}>
        <div className="flex items-center gap-2 mb-1">
          <Zap size={14} style={{ color: 'var(--accent)' }} />
          <span className="font-display text-[14px] font-bold" style={{ color: 'var(--text)' }}>Subscription Plan</span>
          {!isAdmin && (
            <span className="text-[10px] px-2 py-0.5 rounded-full ml-auto" style={{ background: 'rgba(255,107,107,0.1)', color: '#FF6B6B', border: '1px solid rgba(255,107,107,0.2)' }}>
              Admin only
            </span>
          )}
        </div>

        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-3">
          {PLAN_ORDER.map(p => {
            const cfg = PLAN_CONFIGS[p];
            const active = plan === p;
            return (
              <button
                key={p}
                onClick={() => handlePlanChange(p)}
                disabled={!isAdmin}
                className={clsx(
                  'relative flex flex-col gap-2 p-4 rounded-[12px] border text-left transition-all',
                  isAdmin ? 'hover:scale-[1.02] cursor-pointer' : 'cursor-not-allowed opacity-70',
                  active ? 'ring-2' : ''
                )}
                style={{
                  background: active ? `${cfg.color}0d` : 'var(--bg3)',
                  borderColor: active ? cfg.color : 'var(--border)',
                  outline: active ? `2px solid ${cfg.color}` : undefined,
                  outlineOffset: '2px',
                }}
              >
                {active && (
                  <div className="absolute top-2.5 right-2.5 w-5 h-5 rounded-full flex items-center justify-center" style={{ background: cfg.color }}>
                    <Check size={11} color="#fff" />
                  </div>
                )}
                {p === 'pro' && !active && (
                  <div className="absolute top-2.5 right-2.5">
                    <span className="text-[9px] font-bold px-1.5 py-0.5 rounded-full" style={{ background: 'rgba(91,111,255,0.15)', color: 'var(--accent)' }}>Popular</span>
                  </div>
                )}
                <span className="font-display text-[15px] font-bold" style={{ color: active ? cfg.color : 'var(--text)' }}>{cfg.label}</span>
                <span className="text-[13px] font-semibold" style={{ color: 'var(--text2)' }}>{cfg.price}</span>
                <div className="flex flex-wrap gap-1 mt-1">
                  <CapBadge label="Sessions" value={cfg.features.sessionsPerMonth} />
                  <CapBadge label="Agents" value={cfg.features.agentsMax} />
                </div>
              </button>
            );
          })}
        </div>

        {/* Current plan caps summary */}
        <div className="flex flex-wrap gap-3 pt-2 border-t" style={{ borderColor: 'var(--border)' }}>
          <div className="flex items-center gap-2">
            <Clock size={12} style={{ color: 'var(--text3)' }} />
            <span className="text-[11.5px]" style={{ color: 'var(--text2)' }}>
              Session cap: <strong>{features.sessionMinutesMax === null ? 'Unlimited' : `${features.sessionMinutesMax} min`}</strong>
            </span>
          </div>
          <div className="flex items-center gap-2">
            <Users size={12} style={{ color: 'var(--text3)' }} />
            <span className="text-[11.5px]" style={{ color: 'var(--text2)' }}>
              Agents: <strong>{features.agentsMax === null ? 'Unlimited' : features.agentsMax}</strong>
            </span>
          </div>
          <div className="flex items-center gap-2">
            <BarChart3 size={12} style={{ color: 'var(--text3)' }} />
            <span className="text-[11.5px]" style={{ color: 'var(--text2)' }}>
              Sessions/month: <strong>{features.sessionsPerMonth === null ? 'Unlimited' : features.sessionsPerMonth}</strong>
            </span>
          </div>
        </div>
      </div>

      {/* Module toggles */}
      <div className="rounded-[14px] border overflow-hidden" style={{ background: 'var(--bg2)', borderColor: 'var(--border)' }}>
        <div className="px-5 py-4 border-b flex items-center gap-2" style={{ borderColor: 'var(--border)' }}>
          <Shield size={14} style={{ color: 'var(--accent)' }} />
          <span className="font-display text-[14px] font-bold" style={{ color: 'var(--text)' }}>Feature Modules</span>
          <span className="text-[11px] ml-auto" style={{ color: 'var(--text3)' }}>
            {isAdmin ? 'Toggle individual features regardless of plan' : 'Contact an admin to modify modules'}
          </span>
        </div>

        <div className="divide-y" style={{ borderColor: 'var(--border)' }}>
          {MODULES.map((mod, i) => {
            const Icon = mod.icon;
            const enabled = features[mod.key] as boolean;
            const overridden = hasOverride(mod.key);
            const planVal = planDefault(mod.key);

            // Find minimum plan that enables this feature
            const minPlan = PLAN_ORDER.find(p => PLAN_CONFIGS[p].features[mod.key]);
            const locked = !planVal && !overridden;

            return (
              <motion.div
                key={mod.key}
                initial={{ opacity: 0, x: -4 }}
                animate={{ opacity: 1, x: 0 }}
                transition={{ delay: i * 0.03 }}
                className="flex items-center gap-4 px-5 py-4"
              >
                <div
                  className="w-9 h-9 rounded-[10px] flex items-center justify-center flex-shrink-0"
                  style={{
                    background: enabled ? 'rgba(91,111,255,0.1)' : 'var(--bg3)',
                    border: `1px solid ${enabled ? 'rgba(91,111,255,0.2)' : 'var(--border)'}`,
                  }}
                >
                  <Icon size={15} style={{ color: enabled ? 'var(--accent)' : 'var(--text3)' }} />
                </div>

                <div className="flex-1 min-w-0">
                  <div className="flex items-center gap-2 flex-wrap">
                    <span className="text-[13px] font-medium" style={{ color: enabled ? 'var(--text)' : 'var(--text3)' }}>
                      {mod.label}
                    </span>
                    {overridden && (
                      <span className="text-[9px] font-bold uppercase tracking-wide px-1.5 py-0.5 rounded-full" style={{ background: 'rgba(255,209,102,0.15)', color: '#FFD166', border: '1px solid rgba(255,209,102,0.25)' }}>
                        Override
                      </span>
                    )}
                    {!enabled && minPlan && (
                      <span
                        className="text-[9px] font-bold uppercase tracking-wide px-1.5 py-0.5 rounded-full cursor-pointer"
                        style={{ background: `${PLAN_CONFIGS[minPlan].color}18`, color: PLAN_CONFIGS[minPlan].color, border: `1px solid ${PLAN_CONFIGS[minPlan].color}30` }}
                        title={`Available on ${PLAN_CONFIGS[minPlan].label} plan`}
                      >
                        {PLAN_CONFIGS[minPlan].label}+
                      </span>
                    )}
                  </div>
                  <p className="text-[11px] mt-0.5 truncate" style={{ color: 'var(--text3)' }}>{mod.description}</p>
                </div>

                <div className="flex items-center gap-2 flex-shrink-0">
                  {overridden && isAdmin && (
                    <button
                      onClick={() => clearOverride(mod.key)}
                      className="text-[10px] transition-colors hover:text-accent"
                      style={{ color: 'var(--text3)' }}
                      title="Reset to plan default"
                    >
                      Reset
                    </button>
                  )}
                  <button
                    onClick={() => handleToggle(mod.key)}
                    disabled={!isAdmin}
                    className={clsx('transition-all', !isAdmin && 'opacity-40 cursor-not-allowed', isAdmin && 'hover:scale-110 cursor-pointer')}
                    title={enabled ? 'Disable module' : 'Enable module'}
                  >
                    {enabled
                      ? <ToggleRight size={26} style={{ color: 'var(--accent)' }} />
                      : <ToggleLeft size={26} style={{ color: 'var(--text3)' }} />
                    }
                  </button>
                </div>
              </motion.div>
            );
          })}
        </div>
      </div>

      {/* Add-on info */}
      <div className="rounded-[14px] border p-5" style={{ background: 'var(--bg2)', borderColor: 'var(--border)' }}>
        <div className="flex items-center gap-2 mb-3">
          <Star size={14} style={{ color: '#FFD166' }} />
          <span className="font-display text-[14px] font-bold" style={{ color: 'var(--text)' }}>Add-ons</span>
        </div>
        <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
          {[
            { label: 'Knowledge Base', price: '+$49/mo', desc: 'Doc/URL training & pre-call briefings', feature: 'knowledgeBase' as FeatureKey },
            { label: 'Extra Agents', price: '+$9/agent/mo', desc: 'Add agents beyond your plan limit', feature: 'agentsMax' as FeatureKey },
            { label: 'API Access', price: '+$99/mo', desc: 'Full REST API for custom integrations', feature: 'apiAccess' as FeatureKey },
          ].map(addon => (
            <div
              key={addon.label}
              className="rounded-[10px] border p-3.5 flex flex-col gap-2"
              style={{ background: 'var(--bg3)', borderColor: 'var(--border)' }}
            >
              <div className="flex items-start justify-between gap-2">
                <span className="text-[12.5px] font-semibold" style={{ color: 'var(--text)' }}>{addon.label}</span>
                <span className="text-[12px] font-bold flex-shrink-0" style={{ color: '#FFD166' }}>{addon.price}</span>
              </div>
              <p className="text-[11px]" style={{ color: 'var(--text3)' }}>{addon.desc}</p>
              <button
                className="flex items-center gap-1 text-[11px] font-medium transition-colors mt-auto"
                style={{ color: 'var(--accent)' }}
                onClick={() => toast('Contact sales to add this — sales@pitchiq.ai', { icon: '📩' })}
              >
                Contact sales <ChevronRight size={11} />
              </button>
            </div>
          ))}
        </div>
      </div>

    </div>
  );
}
